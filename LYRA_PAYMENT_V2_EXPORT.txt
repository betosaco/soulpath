import { useState, useEffect, useRef, useCallback } from 'react';
import { LyraConfigManager } from '@/lib/lyra/config';

// Define a status to ensure the script is loaded only once per application lifecycle
let scriptStatus: 'idle' | 'loading' | 'ready' | 'error' = 'idle';

interface UseLyraFormProps {
  amount: number;
  currency: string;
  orderId: string;
  customer: { email: string };
  onSuccess?: (data: any) => void;
  onError?: (error: string) => void;
}

export function useLyraForm({
  amount,
  currency,
  orderId,
  customer,
  onSuccess,
  onError,
}: UseLyraFormProps) {
  const [status, setStatus] = useState<'loading' | 'initializing' | 'ready' | 'error'>('loading');
  const [error, setError] = useState<string | null>(null);
  const formContainerRef = useRef<HTMLDivElement>(null);

  // Effect to handle Fast Refresh cleanup in development
  useEffect(() => {
    // Only run cleanup logic in development
    if (process.env.NODE_ENV !== 'development') return;

    // Check if we need to reset script status due to Fast Refresh
    const existingScript = document.querySelector('script[src*="kr-payment-form"]');
    if (existingScript && window.KR && scriptStatus === 'idle') {
      // Script was loaded from previous session, mark as ready
      scriptStatus = 'ready';
    }

    return () => {
      // Cleanup on unmount - only reset in development for Fast Refresh
      if (process.env.NODE_ENV === 'development') {
        // Don't reset scriptStatus here as it might be shared across components
        // Just ensure we don't break existing functionality
      }
    };
  }, []);

  const handleError = useCallback((message: string) => {
    setError(message);
    setStatus('error');
    onError?.(message);
  }, [onError]);

  // Effect 1: Load the Lyra script and CSS once.
  useEffect(() => {
    const lyraConfigInstance = LyraConfigManager.getInstance();
    const config = lyraConfigInstance.getConfig();

    // Log config in development only
    if (process.env.NODE_ENV === 'development') {
      console.log('🔧 Lyra Config:', {
        publicKey: config.publicKey,
        jsLibraryUrl: config.jsLibraryUrl,
        environment: config.environment,
        isProduction: lyraConfigInstance.isProduction(),
        isTest: lyraConfigInstance.isTest()
      });
    }

    // Check if Lyra script is already loaded in the DOM (prevents duplicate loading during Fast Refresh)
    const existingScript = document.querySelector(`script[src="${config.jsLibraryUrl}"]`);
    const existingCss = document.querySelector('link[href*="neon-reset"]');

    if (existingScript && window.KR) {
      // Script is already loaded and KR object exists
      if (scriptStatus !== 'ready') {
        scriptStatus = 'ready';
        setStatus(prev => prev === 'loading' ? 'initializing' : prev);
      }
      return;
    }

    // If script exists but KR is not ready yet, wait for it
    if (existingScript && !window.KR) {
      if (scriptStatus !== 'ready') {
        scriptStatus = 'loading';
        // Wait for KR to be available
        const checkKR = () => {
          if (window.KR) {
            scriptStatus = 'ready';
            setStatus(prev => prev === 'loading' ? 'initializing' : prev);
            window.dispatchEvent(new Event('lyra-script-ready'));
          } else {
            setTimeout(checkKR, 50);
          }
        };
        checkKR();
      }
      return;
    }

    if (scriptStatus === 'ready') {
      return;
    }

    if (scriptStatus === 'idle') {
      scriptStatus = 'loading';

      // Load Neon CSS theme if not already loaded
      if (!existingCss) {
        const neonCss = document.createElement('link');
        neonCss.rel = 'stylesheet';
        neonCss.href = 'https://static.micuentaweb.pe/static/js/krypton-client/V4.0/ext/neon-reset.min.css';
        document.head.appendChild(neonCss);
      }

      // Load Lyra script if not already loaded
      if (!existingScript) {
        const script = document.createElement('script');
        script.src = config.jsLibraryUrl;
        // Format: password:publickey
        const fullPublicKey = `${config.password}:${config.publicKey}`;
        script.setAttribute('kr-public-key', fullPublicKey);
        script.setAttribute('kr-post-url-success', config.returnUrls.success);
        script.setAttribute('kr-language', 'es-ES');
        script.async = true;

        if (process.env.NODE_ENV === 'development') {
          console.log('🔧 Loading Lyra script with public key:', fullPublicKey);
        }

        script.onload = () => {
          scriptStatus = 'ready';
          // Notify all hook instances that the script is ready by triggering a re-render
          window.dispatchEvent(new Event('lyra-script-ready'));
        };

        script.onerror = () => {
          scriptStatus = 'error';
          handleError('Failed to load Lyra payment script.');
        };

        document.head.appendChild(script);
      }
    }

    const onScriptReady = () => setStatus(prev => prev === 'loading' ? 'initializing' : prev);
    window.addEventListener('lyra-script-ready', onScriptReady);

    return () => {
        window.removeEventListener('lyra-script-ready', onScriptReady);
    };
  }, [handleError]);

  // Effect 2: Fetch token and initialize the form.
  useEffect(() => {
    // Wait until script is ready and we have a DOM element
    if (scriptStatus !== 'ready' || !formContainerRef.current) {
      return;
    }

    let isMounted = true;
    let formToken: string | null = null;
    setStatus('initializing');

    const initialize = async () => {
      try {
        // Step 1: Fetch Form Token
        const response = await fetch('/api/lyra/create-token', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ amount, currency, orderId, customer }),
        });

        const data = await response.json();
        if (!data.success || !isMounted) return handleError(data.error || 'Token creation failed.');

        formToken = data.formToken;
        formContainerRef.current!.setAttribute('kr-form-token', formToken!);

        // Step 2: Initialize Lyra Form
        console.log('🎯 Initializing Lyra form...');
        // Enable debug mode only in development/test environments, not in production
        const debugMode = config.environment === 'test' || process.env.NODE_ENV === 'development';
        await window.KR.setDebug(debugMode);

        window.KR.onError((err: any) => {
            console.error('❌ Lyra error:', err);
            const errorMessage = err?.errorMessage || 'An unknown payment error occurred.';
            handleError(errorMessage);
        });

        window.KR.onSuccess((result: any) => {
            console.log('✅ Lyra payment success:', result);
            onSuccess?.(result);
        });

        console.log('🎨 Rendering Lyra elements...');
        await window.KR.renderElements(formContainerRef.current!);
        console.log('✅ Lyra form initialized successfully');
        if (isMounted) setStatus('ready');

      } catch (err) {
        handleError('Failed to initialize payment form.');
      }
    };

    initialize();

    // CRITICAL: Cleanup function to remove the form instance
    return () => {
      isMounted = false;
      if (window.KR && typeof window.KR.removeForms === 'function') {
        try {
          window.KR.removeForms();
        } catch (e) {
          console.error("Error cleaning up Lyra form:", e);
        }
      }
    };
  }, [amount, currency, orderId, customer, onSuccess, handleError]); // Re-initialize if props change

  return { status, error, formContainerRef };
}

  /**

  async validatePayment(krAnswer: string, krHash: string): Promise<LyraValidationResponse> {
    try {
      // Basic validation
      if (!krAnswer || !krHash) {
        return {
          success: false,
          error: 'Missing payment data'

      // Parse the answer to check payment status
      const answerData = JSON.parse(krAnswer);

      const isTestMode = this.config.environment === 'test' || process.env.NODE_ENV === 'development';

      if (answerData.orderStatus === 'PAID') {

          isValid: true,
          data: answerData

        return {

          isValid: false,

  /**

  getLibraryConfig() {
    return {
      jsLibraryUrl: this.config.jsLibraryUrl,
      publicKey: this.config.publicKey
    };
  }

  /**
   * Validates HMAC-SHA-256 signature from Lyra return URL
   * @param data - The data to validate
   * @param signature - The kr-hash signature from Lyra
   * @returns boolean indicating if signature is valid
   */
  validateHmacSignature(data: string, signature: string): boolean {
    try {
      const hmac = crypto.createHmac('sha256', this.config.hmacKey);
      hmac.update(data);
      const calculatedSignature = hmac.digest('base64');
      
      return calculatedSignature === signature;
    } catch (error) {
      console.error('Error validating HMAC signature:', error);
      return false;
    }
  }

  /**
   * Validates payment response from Lyra
   * @param responseData - The response data from Lyra
   * @returns validation result
   */
  validatePaymentResponse(responseData: Record<string, unknown>): { success: boolean; error?: string; data?: Record<string, unknown> } {
    try {
      // Check if required fields are present
      if (!responseData['kr-hash']) {
        return { success: false, error: 'Missing kr-hash signature' };
      }

      if (!responseData['kr-status']) {
        return { success: false, error: 'Missing payment status' };
      }

      // Extract signature and data for validation
      const signature = responseData['kr-hash'];
      const dataForValidation = { ...responseData };
      delete dataForValidation['kr-hash'];
      
      // Create data string for validation (sorted keys)
      const sortedKeys = Object.keys(dataForValidation).sort();
      const dataString = sortedKeys.map(key => `${key}=${JSON.stringify(dataForValidation[key])}`).join('&');

      // Validate signature - but be more lenient in production
      if (this.config.hmacKey && this.config.hmacKey.trim()) {
        if (!this.validateHmacSignature(dataString, signature)) {
          // In production, we might want to be more lenient with signature validation
          // to avoid blocking legitimate webhooks due to signature format differences
        }
      }

      // Check payment status - accept all valid statuses for webhook processing
      const status = responseData['kr-status'];
      const validStatuses = ['PAID', 'CANCELLED', 'REFUSED', 'AUTHORISED', 'CLOSED', 'PENDING', 'FAILED'];

      if (!validStatuses.includes(status)) {
        return {
          success: false,
          error: `Unknown payment status: ${status}`,
          data: responseData
        };
      }

      // For webhook processing, we accept all valid statuses
      // The webhook handler will decide what to do based on the status

      return { 
        success: true, 
        data: {
          ...responseData,
          'kr-hash': signature // Add back the signature for reference
        }
      };

import { useState, useEffect } from 'react';

interface LyraPaymentFormProps {
  amount: number;
  currency: string;
  orderId: string;
  customer: {
    email: string;
    name?: string;
  };
  onSuccess?: (data: any) => void;
  onError?: (error: string) => void;
}

export default function LyraPaymentForm({
  amount,
  currency,
  orderId,
  customer,
  onSuccess,
  onError
}: LyraPaymentFormProps) {
  const [formToken, setFormToken] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // Create form token
  useEffect(() => {
    const createToken = async () => {
      try {
        const response = await fetch('/api/lyra/create-token', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ amount, currency, orderId, customer })
        });

        const data = await response.json();

        if (data.success) {
          setFormToken(data.formToken);
        } else {
          setError(data.error);
          onError?.(data.error);
        }
      } catch (err) {
        const errorMsg = 'Failed to create payment token';
        setError(errorMsg);
        onError?.(errorMsg);
      } finally {
        setLoading(false);
      }
    };

    createToken();
  }, [amount, currency, orderId, customer, onError]);

  // Initialize payment form
  useEffect(() => {
    if (!formToken || typeof window === 'undefined') return;

    // Load Lyra script if not already loaded
    if (!document.querySelector('script[src*="krypton-client"]')) {
      const script = document.createElement('script');
      script.src = 'https://static.micuentaweb.pe/static/js/krypton-client/V4.0/stable/kr-payment-form.min.js';
      script.setAttribute('kr-public-key', 'publickey_UKrWqzlcOvfMEi4OdXuBAcGK1TaTK6izlIJZYWwHGCqkv');
      script.setAttribute('kr-post-url-success', `${window.location.origin}/payment-success`);
      script.setAttribute('kr-language', 'es-ES');

      script.onload = () => {
        console.log('Lyra script loaded');
        initializeForm();
      };

      document.head.appendChild(script);
    } else {
      initializeForm();
    }

    function initializeForm() {
      if (window.KR && formToken) {
        // Set up event handlers
        window.KR.onFormReady(() => {
          console.log('Payment form ready');
          setLoading(false);
        });

        window.KR.onSubmit(() => {
          console.log('Payment submitted');
        });

        window.KR.onError((error: any) => {
          console.error('Payment error:', error);
          setError('Payment failed');
          onError?.('Payment failed');
        });

        // Render form
        setTimeout(() => {
          if (window.KR.renderElements) {
            window.KR.renderElements('#lyra-payment-form');
          }
        }, 100);
      }
    }
  }, [formToken, onError]);

  if (loading) {
    return (
      <div className="flex items-center justify-center p-8">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
        <span className="ml-2">Loading payment form...</span>
      </div>
    );
  }

  if (error) {
    return (
      <div className="p-4 bg-red-50 border border-red-200 rounded">
        <p className="text-red-600">{error}</p>
      </div>
    );
  }

  return (
    <div className="lyra-payment-form-container">
      <div className="kr-embedded" id="lyra-payment-form" kr-form-token={formToken}>
        {/* Form fields will be rendered here by Lyra */}
      </div>
    </div>
  );
}

import { NextRequest, NextResponse } from 'next/server';
import { lyraPaymentService } from '@/lib/lyra/payment-service';

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    
    console.log('🔔 Lyra webhook received:', JSON.stringify(body, null, 2));
    
    // Basic validation - check for required fields
    if (!body['kr-status']) {
      console.error('❌ Missing kr-status in webhook data');
      return NextResponse.json({
        success: false,
        error: 'Missing payment status'
      }, { status: 400 });
    }

    // Try to validate webhook signature, but don't fail if validation fails
    let paymentData = body;
    try {
      const validationResult = lyraPaymentService.validatePaymentResponse(body);
      if (validationResult.success) {
        paymentData = validationResult.data;
        console.log('✅ Webhook signature validated successfully');
      } else {
        console.warn('⚠️ Webhook signature validation failed, but continuing:', validationResult.error);
        // Continue with the original data if validation fails
        paymentData = body;
      }
    } catch (validationError) {
      console.warn('⚠️ Webhook validation error, but continuing:', validationError);
      // Continue with the original data if validation throws an error
      paymentData = body;
    }
    
    // Process the webhook based on payment status
    switch (paymentData['kr-status']) {
      case 'PAID':
        console.log('✅ Payment successful:', paymentData);
        
        // Send confirmation email (with error handling)
        try {
          await sendPaymentConfirmationEmail(paymentData);
        } catch (emailError) {
          console.error('⚠️ Lyra webhook: Failed to send confirmation email:', emailError);
          // Don't fail the webhook if email fails
        }
        break;
        
      case 'AUTHORISED':
        console.log('✅ Payment authorised:', paymentData);
        // TODO: Update order status to authorised
        // Note: Authorised payments may need to be captured later
        break;
        
      case 'CANCELLED':

        break;

async function sendPaymentConfirmationEmail(paymentData: {
  orderId: string;
  amount: number;
  currency: string;
  customerEmail: string;
  customerName: string;
}) {
  try {
    // Extract customer info from payment data
    const customerEmail = paymentData['kr-answer']?.customer?.email || 
                         paymentData.customer?.email || 
                         paymentData.metadata?.customerEmail;
    const customerName = paymentData['kr-answer']?.customer?.name || 
                        paymentData.customer?.name || 
                        paymentData.metadata?.customerName || 
                        'Cliente';
    
    if (!customerEmail) {
      console.warn('⚠️ Lyra webhook: No customer email found in payment data');
      return;
    }

    // Extract package info from metadata
    const packageName = paymentData.metadata?.packageName || 
                       paymentData.metadata?.package_name || 
                       'Paquete de Yoga';
    const amount = paymentData['kr-answer']?.amount || 
                  paymentData.amount || 
                  0;
    const currency = paymentData['kr-answer']?.currency || 
                    paymentData.currency || 
                    'PEN';

    // Try to get email service
    const { createEmailService } = await import('@/lib/brevo-email-service');
    const emailService = await createEmailService();
    
    if (!emailService) {
      console.warn('⚠️ Lyra webhook: Email service not available');
      return;
    }

    // Try to get template from database, fallback to simple email
    let emailContent = '';
    let emailSubject = 'Purchase Confirmation - MatMax Yoga Studio';
    
    try {
      const { CommunicationTemplateService } = await import('@/lib/communication/template-service');
      const template = await CommunicationTemplateService.getTemplate(
        'package_purchase_confirmation',
        'es', // Spanish template
        {
          customer_name: customerName,
          package_name: packageName,
          package_price: (amount / 100).toFixed(2), // Convert from cents
          quantity: '1',
          total_amount: `${currency === 'PEN' ? 'S/.' : currency}${(amount / 100).toFixed(2)}`,
          payment_method: 'Lyra',
          purchase_date: new Date().toLocaleDateString(),
          sessions_count: '1', // Default for now
          session_duration: '60', // Default for now
          expiry_date: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toLocaleDateString(),
          booking_url: `${process.env.NEXT_PUBLIC_APP_URL}/schedule`,
          website_url: process.env.NEXT_PUBLIC_APP_URL || 'https://matmax.world'
        }
      );

      if (template) {
        emailContent = template.content;
        emailSubject = template.subject || emailSubject;
      } else {
        // Fallback to simple email template
        emailContent = createSimpleConfirmationEmail(customerName, packageName, amount, currency);
      }
    } catch (templateError) {
      console.warn('⚠️ Lyra webhook: Could not fetch template, using fallback:', templateError);
      // Fallback to simple email template
      emailContent = createSimpleConfirmationEmail(customerName, packageName, amount, currency);
    }

    // Send the email
    await emailService.sendTemplateEmail(
      customerEmail,
      emailContent,
      emailSubject,
      {}
    );
    
    console.log('✅ Lyra webhook: Purchase confirmation email sent to', customerEmail);
  } catch (error) {
    console.error('⚠️ Lyra webhook: Error in sendPaymentConfirmationEmail:', error);
    throw error; // Re-throw to be caught by the calling function
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';

/**
 * Check Payment Status API
 * 
 * This endpoint allows you to check the current payment status
 * of an order, which is useful for handling cases where IPN
 * might not have been received yet.
 */

export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url);
    const orderId = searchParams.get('orderId');
    const transactionId = searchParams.get('transactionId');

    if (!orderId && !transactionId) {
      return NextResponse.json({
        success: false,
        error: 'Either orderId or transactionId is required'
      }, { status: 400 });
    }

    // Find the order
    const order = await prisma.order.findFirst({
      where: {
        OR: [
          { orderNumber: orderId },
          { id: orderId },
          { transactionId: transactionId }
        ]
      },
      include: {
        payments: true,
        customer: true,
        items: true
      }
    });

    if (!order) {
      return NextResponse.json({
        success: false,
        error: 'Order not found'
      }, { status: 404 });
    }

    // Get the latest payment status
    const latestPayment = order.payments
      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())[0];

    return NextResponse.json({
      success: true,
      order: {
        id: order.id,
        orderNumber: order.orderNumber,
        status: order.status,
        paymentStatus: order.paymentStatus,
        amount: order.total,
        currency: order.currency,
        createdAt: order.createdAt,
        updatedAt: order.updatedAt
      },
      payment: latestPayment ? {
        id: latestPayment.id,
        status: latestPayment.status,
        method: latestPayment.method,
        transactionId: latestPayment.transactionId,
        amount: latestPayment.amount,
        currency: latestPayment.currency,
        processedAt: latestPayment.processedAt,
        failureReason: latestPayment.failureReason
      } : null,
      customer: {
        id: order.customer.id,
        email: order.customer.email,
        firstName: order.customer.firstName,
        lastName: order.customer.lastName
      }
    });

  } catch (error) {
    console.error('❌ Error checking payment status:', error);
    return NextResponse.json({
      success: false,
      error: 'Failed to check payment status'
    }, { status: 500 });
  }
}

export async function POST(request: NextRequest) {
  try {
    const { orderId, transactionId } = await request.json();

    if (!orderId && !transactionId) {
      return NextResponse.json({
        success: false,
        error: 'Either orderId or transactionId is required'
      }, { status: 400 });
    }

    // Find the order
    const order = await prisma.order.findFirst({
      where: {
        OR: [
          { orderNumber: orderId },
          { id: orderId },
          { transactionId: transactionId }
        ]
      },
      include: {
        payments: true,
        customer: true,
        items: true
      }
    });

    if (!order) {
      return NextResponse.json({
        success: false,
        error: 'Order not found'
      }, { status: 404 });
    }

    // If we have a transaction ID, we can query Lyra directly
    if (transactionId) {
      const lyraStatus = await queryLyraPaymentStatus(transactionId);
      
      if (lyraStatus.success) {
        // Update order status based on Lyra response
        await updateOrderFromLyraStatus(order, lyraStatus.data);
      }
    }

    // Return updated order status
    const updatedOrder = await prisma.order.findUnique({
      where: { id: order.id },
      include: {
        payments: true,
        customer: true,
        items: true
      }
    });

    return NextResponse.json({
      success: true,
      order: updatedOrder,
      message: 'Payment status checked and updated if needed'
    });

  } catch (error) {
    console.error('❌ Error checking payment status:', error);
    return NextResponse.json({
      success: false,
      error: 'Failed to check payment status'
    }, { status: 500 });
  }
}

/**
 * Query Lyra API for payment status
 */
async function queryLyraPaymentStatus(transactionId: string) {
  try {
    const username = process.env.LYRA_USERNAME;
    const password = process.env.LYRA_PASSWORD;
    const apiEndpoint = process.env.LYRA_API_ENDPOINT;

    if (!username || !password || !apiEndpoint) {
      throw new Error('Lyra credentials not configured');
    }

    const auth = Buffer.from(`${username}:${password}`).toString('base64');
    
    const response = await fetch(`${apiEndpoint}/GetPaymentDetails`, {
      method: 'POST',
      headers: {
        'Authorization': `Basic ${auth}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        transactionId: transactionId
      })
    });

    const data = await response.json();

    return {
      success: response.ok,
      data: data
    };

  } catch (error) {
    console.error('❌ Error querying Lyra:', error);
    return {
      success: false,
      error: error.message
    };
  }
}

/**
 * Update order status based on Lyra response
 */
async function updateOrderFromLyraStatus(order: any, lyraData: any) {
  try {
    let paymentStatus = 'pending';
    let orderStatus = 'pending';

    switch (lyraData.status) {
      case 'SUCCESS':
        paymentStatus = 'paid';
        orderStatus = 'confirmed';
        break;
      case 'FAILED':
        paymentStatus = 'failed';
        orderStatus = 'cancelled';
        break;
      case 'ABANDONED':
        paymentStatus = 'abandoned';
        orderStatus = 'pending';
        break;
      case 'EXPIRED':
        paymentStatus = 'expired';
        orderStatus = 'cancelled';
        break;
      case 'CANCELLED':
        paymentStatus = 'cancelled';
        orderStatus = 'cancelled';
        break;
    }

    await prisma.order.update({
      where: { id: order.id },
      data: {
        status: orderStatus,
        paymentStatus: paymentStatus,
        transactionId: lyraData.transactionId,
        updatedAt: new Date()
      }
    });

    console.log('✅ Order status updated from Lyra:', {
      orderId: order.id,
      status: orderStatus,
      paymentStatus: paymentStatus
    });

  } catch (error) {
    console.error('❌ Error updating order from Lyra status:', error);
  }
}
